cmake_minimum_required(VERSION 3.22)
project(OpenContinuitySettings VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Svg)
qt_standard_project_setup()
include(GNUInstallDirs)
include(CTest)
add_library(settings-backend STATIC src/DoctorReport.cpp src/StatusModel.cpp src/StatusModel.h src/DoctorProvider.cpp src/DoctorProvider.h src/DeviceInventoryModel.cpp src/DeviceInventoryModel.h src/DeviceInventoryProvider.cpp src/DeviceInventoryProvider.h)
target_include_directories(settings-backend PUBLIC src)
target_link_libraries(settings-backend PUBLIC Qt6::Core)
target_compile_options(settings-backend PRIVATE -Wall -Wextra -Wpedantic -Werror)
function(add_settings_ui target)
    target_link_libraries(${target} PRIVATE settings-backend Qt6::Quick Qt6::QuickControls2 Qt6::Svg)
    qt_add_resources(${target} ui PREFIX "/" FILES
        qml/Main.qml qml/Card.qml qml/SettingsPage.qml assets/opencontinuity-settings.svg)
endfunction()
qt_add_executable(opencontinuity-settings src/main.cpp)
add_settings_ui(opencontinuity-settings)
install(TARGETS opencontinuity-settings RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES assets/opencontinuity-settings.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/opencontinuity-settings.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
if(BUILD_TESTING)
    find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
    qt_add_executable(doctor-fixture tests/fixture.cpp)
    target_link_libraries(doctor-fixture PRIVATE Qt6::Core)
    qt_add_resources(doctor-fixture fixture PREFIX "/" BASE tests FILES tests/report.json)
    qt_add_executable(settings-tests tests/tst_backend.cpp)
    target_link_libraries(settings-tests PRIVATE settings-backend Qt6::Test)
    qt_add_resources(settings-tests fixture PREFIX "/" BASE tests FILES tests/report.json)
    add_test(NAME doctor-backend COMMAND settings-tests $<TARGET_FILE:doctor-fixture>)
    qt_add_executable(inventory-tests tests/tst_inventory.cpp)
    target_link_libraries(inventory-tests PRIVATE settings-backend Qt6::Test)
    add_test(NAME device-inventory-backend COMMAND inventory-tests $<TARGET_FILE:doctor-fixture>)
    qt_add_resources(doctor-fixture inventory-fixture PREFIX "/" BASE tests FILES tests/device-inventory.json)
    qt_add_executable(settings-ui-smoke tests/tst_ui.cpp)
    add_settings_ui(settings-ui-smoke)
    target_link_libraries(settings-ui-smoke PRIVATE Qt6::Test)
    add_test(NAME qml-window-navigation COMMAND settings-ui-smoke)
    set_tests_properties(doctor-backend device-inventory-backend qml-window-navigation PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 30)
endif()
